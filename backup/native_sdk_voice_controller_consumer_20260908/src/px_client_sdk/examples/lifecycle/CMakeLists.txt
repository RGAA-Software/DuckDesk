cmake_minimum_required(VERSION 3.26)
project(PixelsSdkConsumer LANGUAGES C CXX)
include(CTest)

# A consumer supplies only the checkout location (default here is this example's
# own checkout), toolchain and desired mode, never SDK internal include paths.
get_filename_component(default_sdk_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PIXELS_SDK_SOURCE_DIR "${default_sdk_root}" CACHE PATH "Pixels SDK source checkout")
add_subdirectory("${PIXELS_SDK_SOURCE_DIR}" pixels_sdk)
foreach(forbidden IN ITEMS Qt6::Core Qt6::Widgets px_rtc_client px_webrtc_client px_relay_client px_common_win px_common_net px_common_storage)
    if(TARGET ${forbidden})
        message(FATAL_ERROR "SDK-only consumer unexpectedly configured host/RTC target: ${forbidden}")
    endif()
endforeach()
if(PX_SDK_CORE_ONLY AND (TARGET px_sdk_platform OR TARGET px_opus_codec OR FFMPEG_LIBRARIES))
    message(FATAL_ERROR "Transport-only consumer must not configure decoders or codecs")
endif()
add_executable(pixels_sdk_lifecycle main.cpp)
if(PX_SDK_CORE_ONLY)
    target_link_libraries(pixels_sdk_lifecycle PRIVATE pixels::transport)
    target_compile_definitions(pixels_sdk_lifecycle PRIVATE PIXELS_EXAMPLE_CORE_ONLY)
else()
    target_link_libraries(pixels_sdk_lifecycle PRIVATE pixels::sdk)
endif()
if(BUILD_TESTING)
    add_test(NAME pixels_sdk_consumer_lifecycle COMMAND pixels_sdk_lifecycle)
    set_tests_properties(pixels_sdk_consumer_lifecycle PROPERTIES TIMEOUT 20)
endif()
