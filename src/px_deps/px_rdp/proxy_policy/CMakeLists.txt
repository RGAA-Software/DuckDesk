cmake_minimum_required(VERSION 3.25)
project(gammaray_rdp_proxy_policy LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "RDP workspace proxy policy currently supports Windows only")
endif()
set(PX_FREERDP_SOURCE_DIR "" CACHE PATH "Read-only pinned FreeRDP source or SDK headers")
set(PX_FREERDP_BINARY_DIR "" CACHE PATH "Matching FreeRDP generated headers")
if(EXISTS "${PX_FREERDP_SOURCE_DIR}/include/freerdp3/freerdp/server/proxy/proxy_modules_api.h")
    set(_policy_includes "${PX_FREERDP_SOURCE_DIR}/include/freerdp3" "${PX_FREERDP_SOURCE_DIR}/include/winpr3")
    set(_policy_version "${PX_FREERDP_SOURCE_DIR}/include/freerdp3/freerdp/version.h")
    set(_policy_api "${PX_FREERDP_SOURCE_DIR}/include/freerdp3/freerdp/server/proxy/proxy_modules_api.h")
    set(_policy_library "${PX_FREERDP_SOURCE_DIR}/lib/freerdp3.lib")
else()
    set(_policy_includes "${PX_FREERDP_SOURCE_DIR}/include" "${PX_FREERDP_BINARY_DIR}/include"
                         "${PX_FREERDP_SOURCE_DIR}/winpr/include" "${PX_FREERDP_BINARY_DIR}/winpr/include")
    set(_policy_version "${PX_FREERDP_BINARY_DIR}/include/freerdp/version.h")
    set(_policy_api "${PX_FREERDP_SOURCE_DIR}/include/freerdp/server/proxy/proxy_modules_api.h")
    set(_policy_library "${PX_FREERDP_BINARY_DIR}/libfreerdp/Release/freerdp3.lib")
endif()
foreach(required IN ITEMS "${_policy_api}" "${_policy_version}" "${_policy_library}")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "The pinned FreeRDP SDK is required: ${required}")
    endif()
endforeach()
file(READ "${_policy_version}" _policy_version_text)
if(NOT _policy_version_text MATCHES "FREERDP_GIT_REVISION \"aa8650b\"")
    message(FATAL_ERROR "Proxy policy requires the pinned FreeRDP ABI")
endif()
add_library(rdp_proxy_policy SHARED proxy_policy_module.cpp ../rdp_proxy_policy.cpp)
target_compile_features(rdp_proxy_policy PRIVATE cxx_std_23)
target_compile_definitions(rdp_proxy_policy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(rdp_proxy_policy PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ${_policy_includes})
target_link_libraries(rdp_proxy_policy PRIVATE crypt32 "${_policy_library}")
set_target_properties(rdp_proxy_policy PROPERTIES PREFIX "" OUTPUT_NAME "proxy-gammaray-policy-plugin")
