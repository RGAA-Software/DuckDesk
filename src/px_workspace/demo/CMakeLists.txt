cmake_minimum_required(VERSION 3.26)
project(WorkspaceUiDemo LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The workspace UI demo requires Windows")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(duilib CONFIG REQUIRED)
add_executable(workspace_ui_demo WIN32 main.cpp)
target_compile_definitions(workspace_ui_demo PRIVATE UNICODE _UNICODE UILIB_STATIC)
target_compile_options(workspace_ui_demo PRIVATE /utf-8 /W4)
target_link_libraries(workspace_ui_demo PRIVATE dui::duilib gdiplus comctl32 ole32 shell32)
target_link_options(workspace_ui_demo PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'"
    "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/dpi.manifest")
configure_file(workspace.xml workspace.xml COPYONLY)
configure_file("${duilib_DIR}/copyright" duilib-copyright.txt COPYONLY)
enable_testing()
add_test(NAME workspace_ui_demo_lifecycle COMMAND workspace_ui_demo --self-test)
set_tests_properties(workspace_ui_demo_lifecycle PROPERTIES TIMEOUT 30)
foreach(scale IN ITEMS 150 200)
    add_test(NAME workspace_ui_demo_scale_${scale} COMMAND workspace_ui_demo --self-test --scale=${scale})
    set_tests_properties(workspace_ui_demo_scale_${scale} PROPERTIES TIMEOUT 30)
endforeach()
